```typescript
export async function fetchHistory(): Promise<Snapshot[]> {
  const raw = await request<unknown>("/api/portfolio/history");
  const rows = Array.isArray(raw) ? raw : Array.isArray((raw as Record<string, unknown>)?.snapshots)
    ? ((raw as Record<string, unknown>).snapshots as unknown[]) : [];
  return (rows as Record<string, unknown>[]).map((r) => ({
    total_value: num(r.total_value),
    recorded_at: String(r.recorded_at ?? ""),
  }));
}
```

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|--------------|
| client-supplied query params -> SQLite read | `limit`/`since` on `GET /api/portfolio/history` are user/attacker-influenced (any browser can hit this local endpoint) and flow into a parameterized SQL query in `list_snapshots` |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-------------------|
| T-03-05 | Denial of Service | `GET /api/portfolio/history` `limit` param | mitigate | FastAPI `Query(..., ge=1, le=5000)` rejects out-of-range values with 422 before they reach `list_snapshots`, bounding worst-case row count and response size |
| T-03-06 | Tampering | `since` param forwarded into `list_snapshots`'s SQL | accept | `list_snapshots` already parameterizes `since` via `?` placeholders (see `backend/app/db/snapshots.py`), so no new injection surface is introduced by this plan; a malformed non-ISO `since` string simply matches no rows via string comparison, no crash |
</threat_model>
